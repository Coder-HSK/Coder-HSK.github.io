This game runs as a desktop app

Space Invaders uses pygame.sprite groups, real-time rendering at 60fps, and keyboard polling — none of which exist in a browser environment. Run it locally for the full experience with moving aliens, shields, and progressive levels.

How to run it locally

01Install pygame: pip install pygame
02Download space_invaders.py from the GitHub repo below
03Run it: python space_invaders.py
04Arrow keys to move, Space to shoot, R to restart

""" PROJECT 8: Space Invaders Clone ================================= Concepts: pygame sprites, sprite groups, collision detection, enemy movement patterns, shooting mechanics, game states """ import pygame import random, sys WIDTH, HEIGHT = 800, 600 FPS = 60 class Player(pygame.sprite.Sprite): def __init__(self): super().__init__() self.image = pygame.Surface((40, 30), pygame.SRCALPHA) pygame.draw.polygon(self.image, CYAN, [(20,0),(0,28),(40,28)]) self.rect = self.image.get_rect(midbottom=(WIDTH//2, HEIGHT-20)) self.last_shot = 0 def update(self, keys): if keys[pygame.K_LEFT] and self.rect.left > 0: self.rect.x -= PLAYER_SPEED if keys[pygame.K_RIGHT] and self.rect.right < WIDTH: self.rect.x += PLAYER_SPEED def shoot(self): self.last_shot = pygame.time.get_ticks() return Bullet(self.rect.centerx, self.rect.top, -BULLET_SPEED, YELLOW) class Enemy(pygame.sprite.Sprite): COLORS = [RED, (200,80,200), GREEN, CYAN] def __init__(self, col, row): super().__init__() color = Enemy.COLORS[row % len(Enemy.COLORS)] self.image = pygame.Surface((36, 26), pygame.SRCALPHA) pygame.draw.ellipse(self.image, color, (4,8,28,16)) self.rect = self.image.get_rect() self.points = (ENEMY_ROWS - row) * 10 # top rows worth more class Bullet(pygame.sprite.Sprite): def update(self): self.rect.y += self.speed if self.rect.bottom < 0 or self.rect.top > HEIGHT: self.kill() # remove from all sprite groups class Shield(pygame.sprite.Sprite): def hit(self): self.hits -= 1 self.image.set_alpha(int(255 * self.hits / 3)) if self.hits <= 0: self.kill() class Game: def reset(self): self.score = 0; self.lives = 3; self.level = 1 self.all_sprites = pygame.sprite.Group() self.enemies = pygame.sprite.Group() self.player_bullets = pygame.sprite.Group() self.enemy_bullets = pygame.sprite.Group() self.shields = pygame.sprite.Group() def handle_collisions(self): # player bullets vs enemies hits = pygame.sprite.groupcollide( self.enemies, self.player_bullets, True, True) for enemy in hits: self.score += enemy.points # enemy bullets vs player if pygame.sprite.spritecollide(self.player, self.enemy_bullets, True): self.lives -= 1 if self.lives <= 0: self.game_over = True def run(self): while True: for event in pygame.event.get(): if event.type == pygame.QUIT: pygame.quit(); sys.exit() if not self.game_over: keys = pygame.key.get_pressed() self.player.update(keys) if keys[pygame.K_SPACE] and self.player.can_shoot(): b = self.player.shoot() self.player_bullets.add(b) self.handle_collisions() if not self.enemies: self.level += 1 self.spawn_enemies() self.screen.fill(BLACK) self.all_sprites.draw(self.screen) pygame.display.flip() self.clock.tick(FPS) if __name__ == "__main__": Game().run()
